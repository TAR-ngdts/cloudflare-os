export interface NgDotsImportFile {
  path: string;
  kind?: "file" | "symlink";
  size?: number;
}

export interface NgDotsImportRequest {
  files: NgDotsImportFile[];
  packageJson?: {
    dependencies?: Record<string, string>;
    devDependencies?: Record<string, string>;
    scripts?: Record<string, string>;
  };
}

export interface NgDotsSourceFile {
  path: string;
  /** Base64 file content. Paths under .git/ and .github/, secret-bearing files, and symlinks are rejected. */
  contentBase64: string;
  mode?: "100644" | "100755";
}

export interface NgDotsCreateInput {
  businessUnit: string;
  slug: string;
  /** Display name; defaults to the slug. */
  name?: string;
}

export interface NgDotsImportInput extends NgDotsCreateInput {
  /**
   * The source project's files, relative to its root: at most 2000 files, 2 MiB each, 8 MiB total.
   * Vite, Create React App and static sites are supported. Vite and CRA need a root package-lock.json.
   * The source is placed under frontend/; node_modules, dist, build and similar folders are dropped.
   */
  files: NgDotsSourceFile[];
  /** Review-level finding codes (for example "backend_code") the requester accepts. Blocking findings cannot be acknowledged. */
  acknowledgedFindingCodes?: string[];
}

export interface NgDotsOperationStatus {
  approvalId: number;
  operation: "createVibeApp" | "importVibeApp";
  appId: string;
  state: "staged" | "pending" | "running" | "failed" | "completed";
  branch: string;
  stages: {
    repositoryProvisioned: boolean;
    sourceBranchCreated: { branch: string; commitSha: string; files?: number; dropped?: number } | null;
    pullRequestOpened: { number: number; url: string; autoMergeRequested: boolean } | null;
  };
  /** Stages that have not been observed yet. Never assume them. */
  notObserved: string[];
  lastError: string | null;
}

export interface NgDotsSession {
  getIdentity(): Promise<unknown>;
  listApps(): Promise<unknown>;
  listModels(): Promise<unknown>;
  getApp(appId: string): Promise<unknown>;
  inspectImport(request: NgDotsImportRequest): Promise<unknown>;
  /** Queues one approval: provision the repository, publish the pinned NG Dots starter template to a new ng-dots branch, open a pull request. */
  createVibeApp(input: NgDotsCreateInput): Promise<{ approvalId: number }>;
  /** Queues one approval: provision the repository, publish the starter template with this source under frontend/, open a pull request. */
  importVibeApp(input: NgDotsImportInput): Promise<{ approvalId: number }>;
  getVibeAppOperation(approvalId: number): Promise<NgDotsOperationStatus>;
}
